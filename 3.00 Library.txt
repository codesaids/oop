import java.util.Scanner;

class Book
{
String name, author;
boolean issued = false;

static int totalBooks = 0;

Book(String n, String a)
{
name = n;
author = a;
totalBooks++;
}

void display()
{
System.out.println("\nBook Name : " + name);
System.out.println("Author    : " + author);

if(issued)
System.out.println("Status    : Issued");
else
System.out.println("Status    : Available");
}

void issue()
{
if(!issued)
{
issued = true;
System.out.println(name + " issued successfully!");
}
else
{
System.out.println(name + " already issued!");
}
}

void returnBook()
{
if(issued)
{
issued = false;
System.out.println(name + " returned successfully!");
}
else
{
System.out.println(name + " was not issued!");
}
}

public static void main(String[] args)
{
Scanner sc = new Scanner(System.in);

System.out.print("Enter Number of Books: ");
int n = sc.nextInt();
sc.nextLine();

Book b[] = new Book[n];

for(int i = 0; i < n; i++)
{
System.out.print("\nEnter Book Name: ");
String name = sc.nextLine();

System.out.print("Enter Author Name: ");
String author = sc.nextLine();

b[i] = new Book(name, author);
}

int choice, num;

do
{
System.out.println("\n***** LIBRARY MANAGEMENT SYSTEM *****");
System.out.println("1. Display Books");
System.out.println("2. Issue Book");
System.out.println("3. Return Book");
System.out.println("4. Total Books");
System.out.println("5. Exit");

System.out.print("Enter Choice: ");
choice = sc.nextInt();

switch(choice)
{
case 1:
for(int i = 0; i < n; i++)
{
System.out.println("\nBook Number: " + (i + 1));
b[i].display();
}
break;

case 2:
System.out.print("Enter Book Number to Issue: ");
num = sc.nextInt();

if(num >= 1 && num <= n)
b[num - 1].issue();
else
System.out.println("Invalid Book Number!");
break;

case 3:
System.out.print("Enter Book Number to Return: ");
num = sc.nextInt();

if(num >= 1 && num <= n)
b[num - 1].returnBook();
else
System.out.println("Invalid Book Number!");
break;

case 4:
System.out.println("Total Books : " + totalBooks);
break;

case 5:
System.out.println("Thank You!");
break;

default:
System.out.println("Invalid Choice!");
}
}
while(choice != 5);

sc.close();
}
}
