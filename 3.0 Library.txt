import java.util.Scanner;

class Book
{
String bookname;
String author;
boolean isIssued;

static int totalBooks = 0;

public Book(String bookname, String author)
{
this.bookname = bookname;
this.author = author;
this.isIssued = false;
totalBooks++;
}

void displayBook()
{
System.out.println("\nBook Name : " + bookname);
System.out.println("Author    : " + author);

if(isIssued)
{
System.out.println("Status    : Issued");
}
else
{
System.out.println("Status    : Available");
}
}

void issueBook()
{
if(!isIssued)
{
isIssued = true;
System.out.println(bookname + " issued successfully!");
}
else
{
System.out.println(bookname + " already issued!");
}
}

void returnBook()
{
if(isIssued)
{
isIssued = false;
System.out.println(bookname + " returned successfully!");
}
else
{
System.out.println(bookname + " was not issued!");
}
}

public static void main(String[] args)
{
Scanner sc = new Scanner(System.in);

System.out.print("Enter First Book Name: ");
String bname1 = sc.nextLine();

System.out.print("Enter First Book Author: ");
String author1 = sc.nextLine();

System.out.print("Enter Second Book Name: ");
String bname2 = sc.nextLine();

System.out.print("Enter Second Book Author: ");
String author2 = sc.nextLine();

Book b1 = new Book(bname1, author1);
Book b2 = new Book(bname2, author2);

int choice, bookChoice;

do
{
System.out.println("\n***** LIBRARY MANAGEMENT SYSTEM *****");
System.out.println("1. Display Books");
System.out.println("2. Issue Book");
System.out.println("3. Return Book");
System.out.println("4. Total Books");
System.out.println("5. Exit");

System.out.print("Enter choice: ");
choice = sc.nextInt();

switch(choice)
{
case 1:
System.out.println("\n1. " + b1.bookname);
b1.displayBook();

System.out.println("\n2. " + b2.bookname);
b2.displayBook();
break;

case 2:
System.out.println("\nSelect Book to Issue");
System.out.println("1. " + b1.bookname);
System.out.println("2. " + b2.bookname);

System.out.print("Enter Book Number: ");
bookChoice = sc.nextInt();

if(bookChoice == 1)
{
b1.issueBook();
}
else if(bookChoice == 2)
{
b2.issueBook();
}
else
{
System.out.println("Invalid Book Number!");
}
break;

case 3:
System.out.println("\nSelect Book to Return");
System.out.println("1. " + b1.bookname);
System.out.println("2. " + b2.bookname);

System.out.print("Enter Book Number: ");
bookChoice = sc.nextInt();

if(bookChoice == 1)
{
b1.returnBook();
}
else if(bookChoice == 2)
{
b2.returnBook();
}
else
{
System.out.println("Invalid Book Number!");
}
break;

case 4:
System.out.println("Total Books : " + Book.totalBooks);
break;

case 5:
System.out.println("Thank You!");
break;

default:
System.out.println("Invalid Choice!");
}
} 
while(choice != 5);

sc.close();
}
}
